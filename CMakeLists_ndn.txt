# CMakeLists.txt for NDN version of prima.cpp
cmake_minimum_required(VERSION 3.12)
project(prima-ndn)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找NDN-CXX库 (注释掉检查，假设已安装)
# find_package(PkgConfig REQUIRED)
# pkg_check_modules(NDN_CXX REQUIRED libndn-cxx)

# 手动设置NDN-CXX路径 (假设已安装到/usr/local)
set(NDN_CXX_INCLUDE_DIRS "/usr/local/include")
set(NDN_CXX_LIBRARY_DIRS "/usr/local/lib")
set(NDN_CXX_LIBRARIES "ndn-cxx")
set(NDN_CXX_CFLAGS_OTHER "")

# 查找其他依赖
find_package(Threads REQUIRED)

# 包含目录
include_directories(${NDN_CXX_INCLUDE_DIRS})
include_directories(include)
include_directories(common)
include_directories(src)
include_directories(ggml/include)
include_directories(ggml/src)
include_directories(.)

# 编译选项
add_compile_options(${NDN_CXX_CFLAGS_OTHER})

# 定义宏来启用NDN支持
add_definitions(-DUSE_NDN_INSTEAD_OF_ZMQ)

# 添加Linux CPU亲和性支持
add_definitions(-D_GNU_SOURCE)

# 包含构建信息处理
include(cmake/build-info.cmake)

# 生成构建信息文件
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/common/build-info.cpp.in
    ${CMAKE_CURRENT_BINARY_DIR}/build-info.cpp
    @ONLY
)

# GGML源文件
set(GGML_SOURCES
    ggml/src/ggml.c
    ggml/src/ggml-alloc.c
    ggml/src/ggml-backend.cpp
    ggml/src/ggml-quants.c
    ggml/src/llamafile/sgemm.cpp
)

# 源文件
set(PRIMA_SOURCES
    src/llama.cpp
    src/llama-vocab.cpp
    src/llama-grammar.cpp
    src/llama-sampling.cpp
    common/common.cpp
    common/arg.cpp
    common/log.cpp
    common/console.cpp
    common/ngram-cache.cpp
    common/sampling.cpp
    common/train.cpp
    common/profiler.cpp
    # JSON schema to grammar support
    common/json-schema-to-grammar.cpp
    # 添加NDN实现文件
    src/ndn-prima.cpp  # 新的NDN实现文件
    # Unicode支持文件
    src/unicode.cpp
    src/unicode-data.cpp
    # 构建信息文件
    ${CMAKE_CURRENT_BINARY_DIR}/build-info.cpp
    ${GGML_SOURCES}
)

# 创建库
add_library(prima-ndn STATIC ${PRIMA_SOURCES})

# 链接库
target_link_libraries(prima-ndn 
    ${NDN_CXX_LIBRARIES}
    Threads::Threads
)

# 链接目录
target_link_directories(prima-ndn PRIVATE ${NDN_CXX_LIBRARY_DIRS})

# 主程序
add_executable(prima-main-ndn examples/main/main.cpp)
target_link_libraries(prima-main-ndn prima-ndn)

# 编译选项
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(prima-ndn PRIVATE DEBUG_NDN)
endif()

# 性能优化选项
if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(prima-ndn PRIVATE -O3 -DNDEBUG)
endif()

install(TARGETS prima-main-ndn DESTINATION bin)
install(TARGETS prima-ndn DESTINATION lib)
install(DIRECTORY include/ DESTINATION include/prima-ndn)
